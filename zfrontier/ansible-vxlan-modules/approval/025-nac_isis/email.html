# Sample playbook 025 for NaC data model section vxlan.underlay.isis
# Build order: 025 of 129 (2. Underlay)
# Reads ../models/nac_isis.yml with vars_files. Credentials come from the inventory
# placeholders {{ nxos_host }} {{ ansible_user }} {{ ansible_password }} (and {{ nd_host }} for Nexus Dashboard).
# Reference: https://netascode.cisco.com/docs/data_models/vxlan/underlay/underlay_isis/

- name: "025 nac_isis (vxlan.underlay.isis)"
  hosts: ndfc
  gather_facts: false
  vars_files:
    - ../models/nac_isis.yml
  tasks:
    - name: Check the model defines vxlan.underlay.isis
      ansible.builtin.assert:
        that:
          - vxlan['underlay']['isis'] is defined
        quiet: true

    - name: Stage the section into host_vars for the nac_dc_vxlan roles (opt-in with -e stage_model=true)
      ansible.builtin.copy:
        content: "{{ {'vxlan': vxlan} | to_nice_yaml(indent=2) }}"
        dest: "{{ playbook_dir }}/../host_vars/{{ inventory_hostname }}/nac_isis.nac.yaml"
        mode: "0644"
      delegate_to: localhost
      when: stage_model | default(false) | bool

    - name: Validate staged data with cisco.nac_dc_vxlan.validate (run with --tags validate)
      ansible.builtin.import_role:
        name: cisco.nac_dc_vxlan.validate
      tags:
        - never
        - validate
