        dest: "{{ playbook_dir }}/../host_vars/{{ inventory_hostname }}/nac_if_routed_po_interface.nac.yaml"
        mode: "0644"
      delegate_to: localhost
      when: stage_model | default(false) | bool

    - name: Validate staged data with cisco.nac_dc_vxlan.validate (run with --tags validate)
      ansible.builtin.import_role:

        - never
